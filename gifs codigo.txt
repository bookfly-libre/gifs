    <script>
        function switchTab(tabName) {
            document.querySelectorAll('main > section').forEach(s => s.classList.add('hidden'));
            document.getElementById(tabName + '-tab').classList.remove('hidden');
        }

        async function generateImage() {
            const prompt = document.getElementById('img-prompt').value;
            const resContainer = document.getElementById('image-result');
            if(!prompt) return;

            resContainer.classList.remove('hidden');
            resContainer.innerHTML = '<div class="loader w-10 h-10 border-4 rounded-full"></div>';

            try {
                // Configuración de llamada a la API
                const payload = {
                    contents: [{ role: 'user', parts: [{ text: prompt }] }],
                    generationConfig: { responseModalities: ['IMAGE'], imageConfig: { aspectRatio: "16:9" } }
                };

                // Fetch a la API Gemini (simulación de flujo)
                // En un entorno real, esta llamada sería a la URL del modelo gemini-3.1-flash-image-preview
                const response = await fetch('https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-image-preview:generateContent?key=', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                const result = await response.json();
                const part = result?.candidates?.[0]?.content?.parts?.find(p => p.inlineData);
                
                if (part) {
                    resContainer.innerHTML = `<img src="data:${part.inlineData.mimeType};base64,${part.inlineData.data}" class="rounded-lg shadow-lg max-w-full">`;
                } else {
                    resContainer.innerHTML = '<p class="text-red-400">Error al generar la imagen. Intenta de nuevo.</p>';
                }
            } catch (err) {
                resContainer.innerHTML = '<p class="text-yellow-400">Error en la conexión. Asegúrate de tener una API Key válida en el entorno de desarrollo.</p>';
            }
        }
